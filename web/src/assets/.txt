// utils/user.ts
export interface User {
      id: number | string;
        name: string;
          // 其他字段...
}

/**
 * 从双元素数组中排除指定 id 的用户，返回另一个用户
  * @param users 长度为 2 的用户数组
   * @param targetId 要剔除的用户 id
    * @returns 另一个用户对象，若不存在则返回 null
     */
     export function getOtherUser(users: User[], targetId: number | string): User | null {
          const other = users.find(user => user.id !== targetId);
            return other ?? null;
     }

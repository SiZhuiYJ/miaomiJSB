打卡功能网站开发文档

1. 项目概述

1.1 项目目标

开发一个功能完善的打卡网站，支持用户创建打卡计划、每日打卡、图片上传、日历展示等功能，包含完整的用户管理系统和数据分析能力。

1.2 核心功能

· 用户注册登录（邮箱验证）
· 打卡计划创建与管理
· 每日打卡（图片上传+备注）
· 日历视图展示打卡状态
· 补打卡功能
· 暗夜模式切换
· 用户行为埋点
· 单点登录支持
· 多端适配

```typescript
// 1. 图片懒加载和压缩
const useImageOptimization = () => {
  const compressImage = async (file: File, maxSize = 1024 * 1024): Promise<File> => {
    if (file.size <= maxSize) return file

    return new Promise((resolve) => {
      const reader = new FileReader()
      reader.onload = (e) => {
        const img = new Image()
        img.onload = () => {
          const canvas = document.createElement('canvas')
          const ctx = canvas.getContext('2d')!

          let width = img.width
          let height = img.height
          const ratio = Math.sqrt(maxSize / file.size)

          width *= ratio
          height *= ratio

          canvas.width = width
          canvas.height = height
          ctx.drawImage(img, 0, 0, width, height)

          canvas.toBlob((blob) => {
            resolve(new File([blob!], file.name, { type: 'image/jpeg' }))
          }, 'image/jpeg', 0.8)
        }
        img.src = e.target!.result as string
      }
      reader.readAsDataURL(file)
    })
  }
}
```

7. 埋点系统设计

7.1 前端埋点SDK

```typescript
// utils/tracking.ts
class TrackingSDK {
  private queue: TrackingEvent[] = []
  private isSending = false
  private readonly batchSize = 10
  private readonly flushInterval = 5000

  constructor() {
    // 定时上报
    setInterval(() => this.flush(), this.flushInterval)
    // 页面卸载前上报
    window.addEventListener('beforeunload', () => this.flushSync())
  }

  track(event: string, properties: Record<string, any> = {}) {
    const eventData: TrackingEvent = {
      event,
      properties: {
        ...properties,
        url: window.location.href,
        userAgent: navigator.userAgent,
        timestamp: Date.now(),
        sessionId: this.getSessionId(),
        userId: this.getUserId()
      }
    }

    this.queue.push(eventData)

    // 达到批量大小立即上报
    if (this.queue.length >= this.batchSize) {
      this.flush()
    }
  }

  private async flush() {
    if (this.isSending || this.queue.length === 0) return

    this.isSending = true
    const events = this.queue.splice(0, this.batchSize)

    try {
      await fetch('/api/tracking', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ events }),
        keepalive: true // 确保在页面卸载时也能发送
      })
    } catch (error) {
      // 失败重试
      this.queue.unshift(...events)
    } finally {
      this.isSending = false
    }
  }
}

// Vue插件
export const trackingPlugin = {
  install(app: App, options: any) {
    const sdk = new TrackingSDK()
    app.config.globalProperties.$track = sdk.track.bind(sdk)

    // 自动追踪路由
    app.router?.afterEach((to, from) => {
      sdk.track('page_view', {
        page: to.path,
        referrer: from.path
      })
    })
  }
}
```

```csharp
            {
                var accessToken = context.Request.Query["access_token"];
                var path = context.HttpContext.Request.Path;
                if (!string.IsNullOrEmpty(accessToken) && 
                    path.StartsWithSegments("/hubs"))
                {
                    context.Token = accessToken;
                }
                return Task.CompletedTask;
            }
        };
    });

builder.Services.AddScoped<ITokenService, TokenService>();
```
